````
#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Enter n: ";
    cin >> n;

    long long a = 0, b = 1;

    for (int i = 0; i < n; ++i) {
        cout << a << (i + 1 == n ? '\n' : ' ');
        long long next = a + b;
        a = b;
        b = next;
    }

    return 0;
}
````

````
```mermaid
flowchart TD
    A[用户点击 App 图标] --> B[系统创建应用进程]
    B --> C[Application.onCreate]
    C --> D[启动入口 Activity]
    D --> E[Activity.onCreate]
    E --> F[Activity.onStart]
    F --> G[Activity.onResume]
    G --> H[Activity 进入前台，可交互]

    H --> I{用户切到后台?}
    I -- 是 --> J[Activity.onPause]
    J --> K[Activity.onStop]
    K --> L[进入后台]

    L --> M{用户返回 App?}
    M -- 是 --> N[Activity.onRestart]
    N --> F

    I -- 否 --> O{Activity 结束?}
    O -- 是 --> P[Activity.onPause]
    P --> Q[Activity.onStop]
    Q --> R[Activity.onDestroy]

    O -- 否 --> H
````
