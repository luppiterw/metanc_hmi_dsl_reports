bool real_backend_set_offset_rejects_invalid_requests_before_mutation() {

  const auto imported_revision = seed_store(*store, scope, snapshot);
  if (!imported_revision.has_value()) {
    std::filesystem::remove(database_path);
    return false;
  }

  hmi_work_offset::RealWorkOffsetBackend backend(std::move(store), scope);
  const auto table = backend.query_work_offsets();
  const auto* g54 = find_row(table, "G54");
  if (!expect_true(g54 != nullptr, "expected G54 before negative tests")) {
    std::filesystem::remove(database_path);
    return false;
  }
  const auto original_x = g54->offset_x;
  const auto axis_it = g54->component_axis_ids.find("offset_x");
  if (!expect_true(axis_it != g54->component_axis_ids.end(),
                   "expected G54 offset_x axis id")) {
    std::filesystem::remove(database_path);
    return false;
  }

  bool ok = true;
  const auto stale = backend.set_offset({
      .target_ref = g54->target_ref,
      .g_code = g54->g_code,
      .field = "offset_x",
      .axis_id = axis_it->second,
      .value = 42.0,
      .expected_revision = g54->revision - 1,
  });
  ok &= expect_true(!stale.accepted,
                    "expected stale real WorkOffset set_offset to reject");
  ok &= expect_equal(stale.status, "rejected", "stale set_offset status");
  ok &= expect_equal(stale.code,
                     "work_offset.revision_mismatch",
                     "stale set_offset code");

  const auto axis_mismatch = backend.set_offset({
      .target_ref = g54->target_ref,
      .g_code = g54->g_code,
      .field = "offset_x",
      .axis_id = "axis-y",
      .value = 43.0,
      .expected_revision = g54->revision,
  });
  ok &= expect_true(!axis_mismatch.accepted,
                    "expected axis mismatch to reject");
  ok &= expect_equal(axis_mismatch.code,
                     "work_offset.axis_mismatch",
                     "axis mismatch code");

  const auto unknown_field = backend.set_offset({
      .target_ref = g54->target_ref,
      .g_code = g54->g_code,
      .field = "offset_u",
      .axis_id = "axis-u",
      .value = 44.0,
      .expected_revision = g54->revision,
  });
  ok &= expect_true(!unknown_field.accepted,
                    "expected unknown field to reject");
  ok &= expect_equal(unknown_field.code,
                     "work_offset.offset_field_unsupported",
                     "unknown field code");

  const auto unknown_target = backend.set_offset({
      .target_ref = "missing-target",
      .g_code = "G999",
      .field = "offset_x",

      .expected_revision = g54->revision,
