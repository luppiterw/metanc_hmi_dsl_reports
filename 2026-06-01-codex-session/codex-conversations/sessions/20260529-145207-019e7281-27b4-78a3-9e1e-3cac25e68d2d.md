enum class SimLiveAdapterStage {

struct SimLiveSessionCapabilities {

struct SimLiveSession {

  MachineTopology topology;
  SimLiveSessionCapabilities capabilities;

struct SimLiveAdapterOpenResult {

简单说：

auto result = Machine::connect({ .backend = Sim, .endpoint = "profile:demo" });

- 不启动 PLC runtime。
- 不跑真实 scan cycle。
- 不控制 HAL、servo、IO、motion。
- 不做真实 command queue。
- 不让 `/rt-sdk` 直接 include/link `/rt/plc`。
- 不把 `RuntimeCommand`、`CommandResult`、`LocalRuntimeAdapter` 暴露给 SDK 用户。
